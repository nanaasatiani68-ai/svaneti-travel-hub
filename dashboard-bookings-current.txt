"use client";

type Booking = {

  guest_phone: string | null;

  people: number | null;
  total_price: number | null;
  notes: string | null;
  status: string | null;
  created_at: string | null;
  completed_at: string | null;

type Tour = {

  location: string | null;
  organizer_name: string | null;
  contact_phone: string | null;

};

        `áƒ—áƒ¥áƒ•áƒ”áƒœáƒ˜ áƒ¢áƒ£áƒ áƒ”áƒ‘áƒ˜áƒ¡ áƒ©áƒáƒ¢áƒ•áƒ˜áƒ áƒ—áƒ•áƒ áƒ•áƒ”áƒ  áƒ›áƒáƒ®áƒ”áƒ áƒ®áƒ“áƒ: ${ownerToursError.message}`

            tour_id,
            user_id,
            guest_name,
            guest_email,
            guest_phone,
            booking_date,
            people,
            total_price,
            notes,
            status,
            created_at,
            completed_at
          `
        )

        .order("created_at", { ascending: false });

    if (error) {

        `áƒ¯áƒáƒ•áƒ¨áƒœáƒ˜áƒ¡ áƒ¡áƒ¢áƒáƒ¢áƒ£áƒ¡áƒ˜áƒ¡ áƒ¨áƒ”áƒªáƒ•áƒšáƒ áƒ•áƒ”áƒ  áƒ›áƒáƒ®áƒ”áƒ áƒ®áƒ“áƒ: ${error.message}`

    setReceivedBookings((currentBookings) =>
      currentBookings.map((item) =>
        item.id === bookingId

    setSuccessMessage(

        ? "áƒ¯áƒáƒ•áƒ¨áƒáƒœáƒ˜ áƒ¬áƒáƒ áƒ›áƒáƒ¢áƒ”áƒ‘áƒ˜áƒ— áƒ“áƒáƒ“áƒáƒ¡áƒ¢áƒ£áƒ áƒ“áƒ."

  async function cancelMyBooking(booking: PreparedBooking) {
    if (booking.status !== "pending") {
      setErrorMessage(
        "áƒ¨áƒ”áƒ’áƒ˜áƒ«áƒšáƒ˜áƒ áƒ›áƒ®áƒáƒšáƒáƒ“ áƒ›áƒáƒšáƒáƒ“áƒ˜áƒœáƒ¨áƒ˜ áƒ›áƒ§áƒáƒ¤áƒ˜ áƒ¯áƒáƒ•áƒ¨áƒœáƒ˜áƒ¡ áƒ’áƒáƒ£áƒ¥áƒ›áƒ”áƒ‘áƒ."

    setUpdatingId(booking.id);
    setErrorMessage("");
    setSuccessMessage("");

    const { error } = await supabase
      .from("bookings")
      .update({ status: "cancelled" })
      .eq("id", booking.id)
      .eq("user_id", currentUserId)
      .eq("status", "pending");

    if (error) {
      console.error("Booking cancellation error:", error);

      setErrorMessage(
        `áƒ¯áƒáƒ•áƒ¨áƒœáƒ˜áƒ¡ áƒ’áƒáƒ£áƒ¥áƒ›áƒ”áƒ‘áƒ áƒ•áƒ”áƒ  áƒ›áƒáƒ®áƒ”áƒ áƒ®áƒ“áƒ: ${error.message}`
      );

      setUpdatingId(null);
      return;
    }

    setMyBookings((currentBookings) =>
      currentBookings.map((item) =>
        item.id === booking.id
          ? { ...item, status: "cancelled" }
          : item
      )
    );

    setSuccessMessage("áƒ¯áƒáƒ•áƒ¨áƒáƒœáƒ˜ áƒ¬áƒáƒ áƒ›áƒáƒ¢áƒ”áƒ‘áƒ˜áƒ— áƒ’áƒáƒ£áƒ¥áƒ›áƒ“áƒ.");

function normalizePhoneForMessenger(value: string) {
  const trimmedValue = value.trim();
  const hasPlus = trimmedValue.startsWith("+");
  const digits = trimmedValue.replace(/\D/g, "");

  if (!digits) {
    return "";
  }

  return hasPlus ? `+${digits}` : digits;
}

function getWhatsAppUrl(phone: string) {
  const normalizedPhone = normalizePhoneForMessenger(phone).replace(/^\+/, "");

  return normalizedPhone
    ? `https://wa.me/${normalizedPhone}`
    : "https://www.whatsapp.com/";
}

function getViberUrl(phone: string) {
  const normalizedPhone = normalizePhoneForMessenger(phone);

  return normalizedPhone
    ? `viber://chat?number=${encodeURIComponent(normalizedPhone)}`
    : "viber://chat";
}

function formatDate(value: string | null) {
  if (!value) {
    return "áƒáƒ  áƒáƒ áƒ˜áƒ¡ áƒ›áƒ˜áƒ—áƒ˜áƒ—áƒ”áƒ‘áƒ£áƒšáƒ˜";
  }

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    return value;
  }

  return new Intl.DateTimeFormat("ka-GE", {
    year: "numeric",
    month: "long",
    day: "numeric",
    hour: "2-digit",
    minute: "2-digit",
  }).format(date);
}
